cmake_minimum_required(VERSION 3.16)
project(dt46_tools)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio calib3d)
set(DT46_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(record_images src/record_images.cpp)
target_compile_features(record_images PRIVATE cxx_std_20)
ament_target_dependencies(record_images rclcpp sensor_msgs cv_bridge)
target_link_libraries(record_images ${OpenCV_LIBS})
target_include_directories(record_images PRIVATE ${OpenCV_INCLUDE_DIRS})
add_executable(calibrate_camera "${DT46_ROOT}/tools/calibrate_camera.cpp" "${DT46_ROOT}/project/modules/media/calibration.cpp")
find_package(nlohmann_json REQUIRED)
target_compile_features(calibrate_camera PRIVATE cxx_std_20)
target_include_directories(calibrate_camera PRIVATE "${DT46_ROOT}/project" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(calibrate_camera ${OpenCV_LIBS} nlohmann_json::nlohmann_json)
add_executable(make_sentry_variant "${DT46_ROOT}/tools/make_sentry_variant.cpp")
target_compile_features(make_sentry_variant PRIVATE cxx_std_20)
install(TARGETS record_images calibrate_camera make_sentry_variant DESTINATION lib/${PROJECT_NAME})
ament_package()
