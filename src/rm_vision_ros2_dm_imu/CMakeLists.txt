cmake_minimum_required(VERSION 3.16)
project(dm_imu)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(dt46_hardware REQUIRED)
find_package(Threads REQUIRED)
add_executable(dm_imu_node src/node.cpp)
target_compile_features(dm_imu_node PRIVATE cxx_std_20)
target_link_libraries(dm_imu_node dt46_hardware::dt46_hardware Threads::Threads)
ament_target_dependencies(dm_imu_node rclcpp sensor_msgs geometry_msgs)
install(TARGETS dm_imu_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config rviz DESTINATION share/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME} FILES_MATCHING PATTERN "*.xml")
ament_package()
