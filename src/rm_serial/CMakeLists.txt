cmake_minimum_required(VERSION 3.16)
project(rm_serial)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rm_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(dt46_hardware REQUIRED)
find_package(Threads REQUIRED)
add_executable(rm_serial_node src/node.cpp)
target_compile_features(rm_serial_node PRIVATE cxx_std_20)
target_link_libraries(rm_serial_node dt46_hardware::dt46_hardware Threads::Threads)
ament_target_dependencies(rm_serial_node rclcpp rm_interfaces geometry_msgs)
install(TARGETS rm_serial_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME} FILES_MATCHING PATTERN "*.xml")
ament_package()
