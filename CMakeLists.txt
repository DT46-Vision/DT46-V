cmake_minimum_required(VERSION 3.24)
project(dt46_vision VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
include(FetchContent)
include(cmake/hardware.cmake)

if(NOT TARGET nlohmann_json::nlohmann_json)
    find_package(nlohmann_json 3.11 QUIET)
    if(NOT TARGET nlohmann_json::nlohmann_json)
        FetchContent_Declare(json
            URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
            URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d
            TIMEOUT 120 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
        FetchContent_MakeAvailable(json)
    endif()
endif()

# Existing native installations can be supplied through normal CMake cache paths.
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d dnn)
find_path(EIGEN3_INCLUDE_DIR Eigen/Dense PATH_SUFFIXES eigen3 REQUIRED)
add_subdirectory(standalone)

add_library(dt46_pipeline project/modules/vision/pipeline.cpp project/modules/vision/virtual_camera.cpp)
target_include_directories(dt46_pipeline PUBLIC project standalone ${OpenCV_INCLUDE_DIRS})
target_link_libraries(dt46_pipeline PUBLIC dt46_aim nlohmann_json::nlohmann_json ${OpenCV_LIBS})
target_compile_definitions(dt46_pipeline PUBLIC NOMINMAX)
add_library(dt46_transport project/platform/transport.cpp)
target_include_directories(dt46_transport PUBLIC project)
target_link_libraries(dt46_transport PUBLIC nlohmann_json::nlohmann_json)
if(WIN32)
    target_link_libraries(dt46_transport PRIVATE ws2_32)
endif()
target_link_libraries(dt46_pipeline PUBLIC dt46_transport)
include(cmake/graphics.cmake)
include(cmake/media.cmake)
add_executable(dt46_vision project/application/virtual_vision.cpp)
target_link_libraries(dt46_vision PRIVATE dt46_debug dt46_media)
target_compile_definitions(dt46_vision PRIVATE DT46_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
if(BUILD_TESTING)
    add_executable(dt46_hardware_check tests/hardware_check.cpp)
    target_link_libraries(dt46_hardware_check PRIVATE dt46_hardware nlohmann_json::nlohmann_json)
    add_test(NAME hardware_protocol COMMAND dt46_hardware_check "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/protocol_reference.json")
    add_executable(dt46_pipeline_check tests/pipeline_check.cpp)
    target_link_libraries(dt46_pipeline_check PRIVATE dt46_pipeline)
    add_test(NAME native_pipeline COMMAND dt46_pipeline_check "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(dt46_transport_check tests/transport_check.cpp)
    target_link_libraries(dt46_transport_check PRIVATE dt46_transport)
    add_test(NAME native_transport COMMAND dt46_transport_check)
endif()
